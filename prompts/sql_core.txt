RULE 1 — AGGREGATION IS MANDATORY (NEVER SKIP)
═══════════════════════════════════════════

RULE 2 — KPI FORMULAS (USE EXACTLY AS WRITTEN)
═══════════════════════════════════════════

── EGA % ──────────────────────────────────
((SUM(t_weight) - SUM(theoretical_pack_weight)) / NULLIF(SUM(t_weight), 0)) * 100 AS ega_percent
Table: itciot.ega_details_data
⚠ NEVER use AVG(ega_percent) for multi-hour aggregation. Always re-derive from t_weight.

── OEE % ──────────────────────────────────
(
  ((SUM(good_bags) / NULLIF((SUM(1)*60 - SUM(downtime_mins)), 0)) / NULLIF(AVG(target_speed)/60.0, 0) * 100)
  * ((SUM(1)*60 - SUM(downtime_mins)) / NULLIF(SUM(1)*60, 0) * 100)
  * (CASE WHEN SUM(good_bags)=0 THEN 0
     ELSE 100 - ((SUM(overlimit_count)+SUM(failed_bags)) / NULLIF(SUM(good_bags),0)) * 100
     END)
) / 10000 AS oee_percent
Table: itciot.oee_details_data
Note: SUM(1) = row count (each row = 1 hour). target_speed is bags/hour → divide by 60 for bags/min.

── Availability % ─────────────────────────
((SUM(1)*60 - SUM(downtime_mins)) / NULLIF(SUM(1)*60, 0)) * 100 AS availability_pct
Table: itciot.oee_details_data

── Performance % ──────────────────────────
((SUM(good_bags) / NULLIF(SUM(1)*60 - SUM(downtime_mins), 0)) * 60)
  / NULLIF(AVG(target_speed), 0) * 100 AS performance_pct
Table: itciot.oee_details_data
Note: multiply actual speed by 60 to convert bags/min → bags/hr before dividing by target_speed (bags/hr).

── Quality % ──────────────────────────────
CASE WHEN SUM(good_bags)=0 THEN 0
ELSE 100 - ((SUM(overlimit_count)+SUM(failed_bags)) / NULLIF(SUM(good_bags),0)) * 100
END AS quality_pct
Table: itciot.oee_details_data
⚠ NEVER omit overlimit_count. Using only failed_bags overstates quality.

── Downtime ───────────────────────────────
SUM(downtime_mins) AS total_downtime
Table: itciot.oee_details_data

── MTTR (mins) ────────────────────────────
CASE WHEN (SUM(jaw_jam_error)+SUM(jam_error)+SUM(fmd_error)+SUM(splice_error)+SUM(dump_error)+SUM(print_check_err)+SUM(reg_mark_err))=0
THEN 0
ELSE SUM(downtime_mins) / (SUM(jaw_jam_error)+SUM(jam_error)+SUM(fmd_error)+SUM(splice_error)+SUM(dump_error)+SUM(print_check_err)+SUM(reg_mark_err))
END AS mttr_mins
Table: itciot.oee_details_data

── MTBF (mins) ────────────────────────────
CASE WHEN (SUM(jaw_jam_error)+SUM(jam_error)+SUM(fmd_error)+SUM(splice_error)+SUM(dump_error)+SUM(print_check_err)+SUM(reg_mark_err))=0
THEN SUM(run_duration)
ELSE SUM(run_duration) / (SUM(jaw_jam_error)+SUM(jam_error)+SUM(fmd_error)+SUM(splice_error)+SUM(dump_error)+SUM(print_check_err)+SUM(reg_mark_err))
END AS mtbf_mins
Table: itciot.oee_details_data

── Total Production (kg) ──────────────────
SUM(COALESCE(good_bags,0) * COALESCE(grammage,0)) / 1000 AS total_production_kg
Table: itciot.oee_details_data

── Laminate Wastage (kg) ──────────────────
SUM(laminate_wastage) AS laminate_wastage_kg
Table: itciot.oee_details_data

── Manual Wastage (kg) ────────────────────
SUM(wastage_kg) AS manual_wastage_kg
Table: itciot.wastage_records
Date column: production_start_time (NOT start_time)

── Production Speed (BPM) ─────────────────
SELECT AVG(value::numeric) AS speed_bpm
FROM itciot.production_speed_details_data,
     jsonb_array_elements(production_speed_bpm) AS value
WHERE machine_id = N AND start_time::date = 'YYYY-MM-DD'
⚠ Use jsonb_array_elements (NOT jsonb_array_elements_text). production_speed_bpm is jsonb.

── Net Efficiency % ───────────────────────
SELECT (SUM((speed.value)::numeric) / NULLIF(SUM(oee.target_speed), 0)) * 100 AS net_efficiency_pct
FROM itciot.production_speed_details_data psd
CROSS JOIN LATERAL jsonb_array_elements(psd.production_speed_bpm) AS speed(value)
JOIN itciot.oee_details_data oee ON psd.machine_id = oee.machine_id AND psd.start_time = oee.start_time

── GSM Vendor Consumption ─────────────────
SELECT vendor,
  GREATEST(0,
    SUM(CASE WHEN weight_type='Starting Weight' THEN weight ELSE 0 END) -
    SUM(CASE WHEN weight_type='Ending Weight' THEN weight ELSE 0 END)
  ) AS gsm_consumption
FROM itciot.gsm_usage_details GROUP BY vendor

RULE 3 — DATE & TIME FILTERS
═══════════════════════════════════════════
• ega / oee / speed tables     → start_time::date = 'YYYY-MM-DD'
• wastage_records               → production_start_time::date = 'YYYY-MM-DD'
• "last 7 days"                 → start_time >= CURRENT_DATE - INTERVAL '7 days'
• "this week"                   → start_time >= date_trunc('week', CURRENT_DATE)
• "May 2025"                    → EXTRACT(MONTH FROM start_time)=5 AND EXTRACT(YEAR FROM start_time)=2025
• "between A and B"             → start_time >= 'A' AND start_time <= 'B'

Shift time ranges (apply to start_time when oee/ega table; use shift column for wastage_records):
  Shift 1 / Morning → start_time::time >= '06:00' AND start_time::time < '14:00'
  Shift 2 / Afternoon → start_time::time >= '14:00' AND start_time::time < '22:00'
  Shift 3 / Night → start_time::time >= '22:00' OR start_time::time < '06:00'
  wastage_records shift column: use ILIKE '%1%' / '%2%' / '%3%' or '%morning%' / '%night%'

Q: OEE for all machines on YYYY-MM-DD
SELECT machine_id, machine_name,
  ((
    ((SUM(good_bags)/NULLIF((SUM(1)*60-SUM(downtime_mins)),0))/NULLIF(AVG(target_speed)/60.0,0)*100)
    * ((SUM(1)*60-SUM(downtime_mins))/NULLIF(SUM(1)*60,0)*100)
    * (CASE WHEN SUM(good_bags)=0 THEN 0
       ELSE 100-((SUM(overlimit_count)+SUM(failed_bags))/NULLIF(SUM(good_bags),0))*100 END)
  ) / 10000) AS oee_percent
FROM itciot.oee_details_data
WHERE start_time::date = 'YYYY-MM-DD'
GROUP BY machine_id, machine_name
ORDER BY oee_percent DESC

Q: Average speed for machine 3 on YYYY-MM-DD

Q: Total wastage by plant name
SELECT p.plant_name, SUM(w.wastage_kg) AS total_wastage_kg
FROM itciot.wastage_records w
JOIN gmiiot_plants p ON w.plant_id = p.plant_id
WHERE p.plant_name ILIKE '%Plant 1%'
GROUP BY p.plant_name

Q: Quality percent for all machines last 7 days
SELECT machine_id, machine_name,
  CASE WHEN SUM(good_bags)=0 THEN 0
  ELSE 100-((SUM(overlimit_count)+SUM(failed_bags))/NULLIF(SUM(good_bags),0))*100
  END AS quality_pct
FROM itciot.oee_details_data
WHERE start_time >= CURRENT_DATE - INTERVAL '7 days'
GROUP BY machine_id, machine_name
ORDER BY quality_pct DESC

Q: GSM consumption by vendor

Q: MTTR for machine 5 this month
SELECT machine_id,
  CASE WHEN (SUM(jaw_jam_error)+SUM(jam_error)+SUM(fmd_error)+SUM(splice_error)+SUM(dump_error)+SUM(print_check_err)+SUM(reg_mark_err))=0
  THEN 0
  ELSE SUM(downtime_mins)/(SUM(jaw_jam_error)+SUM(jam_error)+SUM(fmd_error)+SUM(splice_error)+SUM(dump_error)+SUM(print_check_err)+SUM(reg_mark_err))

FROM itciot.oee_details_data
WHERE machine_id = 5
  AND EXTRACT(MONTH FROM start_time) = EXTRACT(MONTH FROM CURRENT_DATE)
  AND EXTRACT(YEAR FROM start_time) = EXTRACT(YEAR FROM CURRENT_DATE)
GROUP BY machine_id
